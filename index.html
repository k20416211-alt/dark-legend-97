// DARK LEGEND 97 - PLAYLIST PERSISTENCE const DB_NAME = 'dark-legend-97-database'; const DB_VERSION = 1; const STORE_NAME = 'savedSongs'; function openPlaylistDB() { return new Promise((resolve, reject) => { const request = indexedDB.open(DB_NAME, DB_VERSION); request.onupgradeneeded = () => { const db = request.result; if (!db.objectStoreNames.contains(STORE_NAME)) { db.createObjectStore(STORE_NAME, { keyPath: 'id' }); } }; request.onsuccess = () => resolve(request.result); request.onerror = () => reject(request.error); }); } async function saveSongToDB(song) { try { const db = await openPlaylistDB(); const record = { id: song.id, name: song.name, url: song.url || null, file: song.file || null, order: songs.findIndex(item => item.id === song.id) }; return new Promise((resolve, reject) => { const tx = db.transaction(STORE_NAME, 'readwrite'); tx.objectStore(STORE_NAME).put(record); tx.oncomplete = () => { db.close(); resolve(); }; tx.onerror = () => { db.close(); reject(tx.error); }; }); } catch (error) { console.error('Playlist save error:', error); say('A lejátszási lista mentése nem sikerült.'); } } async function restorePlaylist() { try { const db = await openPlaylistDB(); const records = await new Promise((resolve, reject) => { const tx = db.transaction(STORE_NAME, 'readonly'); const request = tx.objectStore(STORE_NAME).getAll(); request.onsuccess = () => resolve(request.result); request.onerror = () => reject(request.error); }); db.close(); records.sort((a, b) => a.order - b.order); songs = records.map(item => ({ id: item.id, name: item.name, url: item.url, file: item.file })); renderList(); if (songs.length) { say(`${songs.length} korábban mentett zene visszatöltve.`); } } catch (error) { console.error('Playlist restore error:', error); say('A mentett lista betöltése nem sikerült.'); } } async function clearSavedPlaylist() { try { const db = await openPlaylistDB(); await new Promise((resolve, reject) => { const tx = db.transaction(STORE_NAME, 'readwrite'); tx.objectStore(STORE_NAME).clear(); tx.oncomplete = resolve; tx.onerror = () => reject(tx.error); }); db.close(); } catch (error) { console.error('Playlist clear error:', error); } }
